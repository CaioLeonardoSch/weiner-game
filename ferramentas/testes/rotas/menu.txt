# fase de raça fixa (N09, Border Collie) ignora a escolha; concluir guarda a raça; a pausa e a
# região nova levam à tela "antes de jogar".
esperar 30
checar jogo.get_node("/root/Fases").raca_da_regiao(&"floresta").id == &"salsicha"
chamar res://ferramentas/testes/teste_menu.gd abrir_floresta

checar jogo._titulo_tela.text == "Floresta"
checar jogo._tela.find_child("Raca_pug", true, false) != null
checar jogo._tela.find_child("Raca_border_collie", true, false) == null
chamar res://ferramentas/testes/teste_menu.gd escolher_pug
esperar 5
checar jogo.get_node("/root/Fases").raca_da_regiao(&"floresta").id == &"pug"

chamar res://ferramentas/testes/teste_menu.gd jogar_f01

# "Trocar de raça" (pausa) / região nova: o menu abre na tela antes de jogar.

checar jogo._titulo_tela.text == "Floresta"
checar jogo._tela.find_child("Jogar", true, false) != null
