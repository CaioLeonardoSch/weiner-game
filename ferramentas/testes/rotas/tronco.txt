# O tronco: empurrado de lado rola, ao comprido desliza; mordendo a ponta e andando para trás, é
# puxado; na correnteza boia, desce e, encalhado na margem, volta para o lugar. O graveto ao
# comprido empurra de longe; o tronco caído como pinguela é passagem estreita; a ponte velha
# cede com alguém parado em cima. O tronco e o graveto comprido acionam a placa com qualquer parte.
fase res://scenes/fases/fase_11.tscn
esperar 10
chamar res://ferramentas/testes/teste_tronco.gd preparar_tronco
esperar 5
# De lado (-Z): rola uma fileira.
apertar mover_frente 30
esperar 50
checar jogo.get_meta("tronco").celulas() == [Vector3i(1, 0, -4), Vector3i(2, 0, -4)]
# Ao comprido (+X): desliza uma célula.
chamar res://ferramentas/testes/teste_tronco.gd cachorro_na_ponta_oeste
esperar 5
apertar mover_direita 30
esperar 50
checar jogo.get_meta("tronco").celulas() == [Vector3i(2, 0, -4), Vector3i(3, 0, -4)]
# Mordendo a ponta oeste (F de frente para ela) e andando para trás (-X): vem junto.
chamar res://ferramentas/testes/teste_tronco.gd cachorro_na_ponta_oeste
esperar 5
checar jogo.get_meta("tronco").ponta_perto(jogo.cachorro.global_position) == 0
apertar acao 40
esperar 3
apertar mover_esquerda 12
esperar 40
checar jogo.get_meta("tronco").celulas() == [Vector3i(1, 0, -4), Vector3i(2, 0, -4)]
checar jogo.cachorro.global_position.x < 1.0
# Na correnteza: boia, desce o rio e encalha na margem (onde o rio estreita) — volta ao lugar.
chamar res://ferramentas/testes/teste_tronco.gd preparar_encalhe
esperar 5
checar jogo.get_meta("encalhe").estado == 1
esperar 360
checar jogo.get_meta("voltas") >= 1
# Graveto comprido ao comprido: a ponta empurra o bloco de longe (o cachorro fica a mais de
# 1 m dele).
fase res://scenes/fases/fase_06.tscn
esperar 10
chamar res://ferramentas/testes/teste_tronco.gd preparar_graveto_empurra
chamar res://ferramentas/testes/teste_tronco.gd recuar_do_bloco
esperar 5
chamar res://ferramentas/testes/teste_tronco.gd virar
esperar 20
checar jogo.cachorro.graveto_ao_comprido
apertar mover_direita 40
esperar 60
checar jogo.get_meta("bloco").global_position.x > 5.0
checar jogo.cachorro.global_position.x < jogo.get_meta("bloco").global_position.x - 1.0
# Tronco caído como pinguela sobre o canal fundo: passagem estreita.
chamar res://ferramentas/testes/teste_tronco.gd preparar_pinguela
esperar 5
checar not jogo.fase.passagem_estreita_em(Vector3(11.5, 0.1, -4.5)).is_empty()
checar jogo.fase.passagem_estreita_em(Vector3(11.5, 0.1, -5.5)).is_empty()
# Tronco na placa (pedra e madeira): aciona com qualquer célula em cima — o meio, a ponta 2 e a
# ponta 0 —, e solta quando nenhuma está. O graveto comprido aciona só com a ponta em cima.
chamar res://ferramentas/testes/teste_tronco.gd preparar_tronco_na_placa
esperar 10
checar jogo.get_meta("placa_pedra").ativa and jogo.get_meta("placa_madeira").ativa
chamar res://ferramentas/testes/teste_tronco.gd troncos_com_a_ponta_2
esperar 5
checar jogo.get_meta("placa_pedra").ativa and jogo.get_meta("placa_madeira").ativa
chamar res://ferramentas/testes/teste_tronco.gd troncos_com_a_ponta_0
esperar 5
checar jogo.get_meta("placa_pedra").ativa and jogo.get_meta("placa_madeira").ativa
chamar res://ferramentas/testes/teste_tronco.gd troncos_fora_das_placas
esperar 5
checar not jogo.get_meta("placa_pedra").ativa and not jogo.get_meta("placa_madeira").ativa
chamar res://ferramentas/testes/teste_tronco.gd graveto_com_a_ponta_na_placa
esperar 5
checar jogo.get_meta("placa_madeira").ativa and not jogo.get_meta("placa_pedra").ativa
# Ponte que cede: passar correndo não quebra; parado em cima, quebra e o cachorro cai na água e
# volta para a margem.
fase res://scenes/fases/fase_08.tscn
esperar 10
chamar res://ferramentas/testes/teste_tronco.gd preparar_ponte_que_cede
esperar 5
apertar mover_direita 50
esperar 60
checar jogo.cachorro.global_position.x > 9.5
checar not jogo.get_meta("ponte").quebrada
apertar mover_esquerda 22
esperar 90
checar jogo.get_meta("ponte").quebrada
esperar 60
checar jogo.cachorro.global_position.y > -0.1
checar absf(jogo.cachorro.global_position.x - 8.5) > 0.6
