# F02 — A outra perspectiva: o barranco (2 m) tem um túnel que a câmera isométrica não mostra.
# Na ida o túnel não serve (na saída leste há um degrau de meio bloco): sobe pela rampa dupla e
# desce do outro lado. No mirante vê a volta; com o graveto (3D) volta pelo túnel.
fase res://scenes/fases/floresta_02_perspectiva.tscn
esperar 20
# Pelo túnel, na ida, não passa.
ir 5.5 -2.5
ir 10.5 -2.5 150
checar jogo.cachorro.global_position.x < 10.0 and jogo.cachorro.global_position.y < 0.3
ir 5.5 -2.5
# Mirante: F olha a fase como ela fica na volta; Esc sai.
ir 2.5 -2.0
apertar mover_tras 6
esperar 10
apertar acao 2
esperar 90
checar jogo._mirante != null
apertar liberar_mouse 2
esperar 120
checar jogo._mirante == null
# Pela rampa dupla, em cima do barranco, e desce do outro lado.
ir 1.5 -5.5
ir 7.5 -5.5
checar jogo.cachorro.global_position.y > 1.9
ir 10.5 -5.5
esperar 30
checar jogo.cachorro.global_position.y < 0.7
ir 12.5 -3.5
esperar 10
checar jogo.cachorro.tem_graveto
# Volta (3D): pelo túnel, descendo o meio bloco.
ir 10.0 -2.5
ir 5.5 -2.5
checar jogo.cachorro.global_position.x < 6.0
checar jogo.cachorro.global_position.y < 0.3
ir -2.5 -3.0
esperar 60
checar jogo.concluida
