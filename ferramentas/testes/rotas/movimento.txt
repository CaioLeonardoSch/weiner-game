# Etapa 4 — rampa lisa (com graveto pesado o cachorro escorrega e não sobe) e degrau alto
# (0,72 m: só pulando, e com graveto na boca o pulo não chega).
fase res://ferramentas/testes/cenas/campo.tscn
esperar 10
chamar res://ferramentas/testes/teste_movimento.gd preparar
# Rampa lisa com graveto pesado: não sobe.
chamar res://ferramentas/testes/teste_movimento.gd antes_da_rampa
chamar res://ferramentas/testes/teste_movimento.gd graveto_pesado
esperar 5
apertar mover_direita 90
esperar 95
checar jogo.cachorro.global_position.x < 2.6 and jogo.cachorro.global_position.y < 0.3
# Sem graveto: sobe.
chamar res://ferramentas/testes/teste_movimento.gd tirar_graveto
chamar res://ferramentas/testes/teste_movimento.gd antes_da_rampa
esperar 5
apertar mover_direita 40
esperar 45
checar jogo.cachorro.global_position.x > 3.0 and jogo.cachorro.global_position.y > 0.4
# Degrau alto: sem graveto, pulando, sobe.
chamar res://ferramentas/testes/teste_movimento.gd antes_do_degrau
esperar 5
apertar mover_direita 40
esperar 12
apertar pular 3
esperar 40
checar jogo.cachorro.global_position.y > 0.65
# Com graveto leve na boca: não.
chamar res://ferramentas/testes/teste_movimento.gd antes_do_degrau
chamar res://ferramentas/testes/teste_movimento.gd graveto_leve
esperar 5
apertar mover_direita 40
esperar 12
apertar pular 3
esperar 90
checar jogo.cachorro.global_position.y < 0.3
