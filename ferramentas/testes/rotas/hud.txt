# HUD e controles (issue #13): barra de equilíbrio escondida no começo; R reinicia; puxar vira
# até agarrar e, sem espaço atrás, não puxa (sem texto na tela); a correnteza não leva o
# cachorro parado à força.
fase res://ferramentas/testes/cenas/ponte_velha.tscn

checar not jogo.indicador_equilibrio.visible

apertar reiniciar 2

checar jogo.fase != null and not jogo.get_tree().paused
# Puxar contra o portão: vira para o bloco segurando F (chegando de lado); sem espaço atrás,
# não puxa e não aparece aviso.
fase res://ferramentas/testes/cenas/campo.tscn
camera 0
esperar 20
chamar res://ferramentas/testes/teste_hud.gd puxar_contra_o_portao
esperar 5
apertar acao 70

esperar 40

# Correnteza (Fase 04): parado à força, a água não leva o cachorro.
fase res://ferramentas/testes/cenas/riacho.tscn
camera 0
esperar 20
teleportar 4.5 0.3 -10.5
esperar 10
# Solto, a correnteza leva o cachorro (confere que ele está mesmo na correnteza)...
eval jogo.set_meta("z", jogo.cachorro.global_position.z)
esperar 30
checar jogo.cachorro.global_position.z - jogo.get_meta("z") > 0.2
# ... parado à força, não.
eval jogo.cachorro.set("entrada_bloqueada", true)
eval jogo.set_meta("z", jogo.cachorro.global_position.z)
esperar 60
checar absf(jogo.cachorro.global_position.z - jogo.get_meta("z")) < 0.05
